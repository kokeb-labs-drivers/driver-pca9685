205template <typename Port>
206[[nodiscard]] Error TransferError(Port& port) noexcept {
207 if constexpr (HasGetLastFailure<Port>) {
208 I2cFailure failure = I2cFailure::Other;
209 if (port.GetLastFailure(failure) && (failure == I2cFailure::AddressNack)) {
210 return Error::NoDevice;
211 }
212 } else {
213 (void)port;
214 }
215 return Error::Bus;
216}

252 Error Read(uint8_t reg, std::span<uint8_t> out) noexcept {
253 if (out.empty() || (out.size() > kMaxBurst)) {
254 return Error::InvalidArgument;
255 }
256 const std::array<uint8_t, 1> pointer{reg};
257 return port_.WriteRead(address_, pointer.data(), pointer.size(), out.data(), out.size())
258 ? Error::Ok
259 : TransferError(port_);
260 }

268
271 Error Write(uint8_t reg, std::span<const uint8_t> data) noexcept {
272 if (data.empty() || (data.size() > kMaxBurst)) {
273 return Error::InvalidArgument;
274 }
275 std::array<uint8_t, 1U + kMaxBurst> frame{};
276 frame[0] = reg;
277 const std::size_t count = data.size();
278 for (std::size_t i = 0U; i < count; ++i) {
279 frame[1U + i] = data[i];
280 }
281 return port_.Write(address_, frame.data(), count + 1U) ? Error::Ok : TransferError(port_);
282 }
283
284private:
285 Port& port_;
286 uint8_t address_;
287};
