38[[nodiscard]] constexpr std::array<uint8_t, 4> EncodePwm(const Pwm& pwm) noexcept {
39 const auto on = static_cast<uint32_t>(pwm.on_count) & 0x0FFFU;
40 const auto off = static_cast<uint32_t>(pwm.off_count) & 0x0FFFU;
41 const uint32_t on_full = pwm.full_on ? static_cast<uint32_t>(reg::kLedFull) : 0U;
42 const uint32_t off_full = pwm.full_off ? static_cast<uint32_t>(reg::kLedFull) : 0U;
43 return {static_cast<uint8_t>(on & 0xFFU), static_cast<uint8_t>((on >> 8U) | on_full),
44 static_cast<uint8_t>(off & 0xFFU), static_cast<uint8_t>((off >> 8U) | off_full)};
45}

52[[nodiscard]] constexpr Result<Pwm> DecodePwm(std::span<const uint8_t, 4> b) noexcept {
53 const auto on_h = static_cast<uint32_t>(b[1]);
54 const auto off_h = static_cast<uint32_t>(b[3]);
55 const auto reserved = static_cast<uint32_t>(reg::kLedHReserved);
56 if (((on_h & reserved) != 0U) || ((off_h & reserved) != 0U)) {
57 return Result<Pwm>::Failure(Error::InvalidReply);
58 }
59 const auto high = static_cast<uint32_t>(reg::kLedCountHigh);
60 const auto full = static_cast<uint32_t>(reg::kLedFull);
61 return Result<Pwm>::Success(Pwm{.on_count = static_cast<uint16_t>(((on_h & high) << 8U) | b[0]),
62 .off_count = static_cast<uint16_t>(((off_h & high) << 8U) | b[2]),
63 .full_on = (on_h & full) != 0U,
64 .full_off = (off_h & full) != 0U});
65}

70[[nodiscard]] constexpr uint32_t OffLevelOutne(const Config& c) noexcept {

78[[nodiscard]] constexpr uint8_t Mode2Value(const Config& c) noexcept {
79 uint32_t v = 0U;
80 if (c.invert_outputs) {
81 v |= static_cast<uint32_t>(reg::kMode2Invrt);
82 }
83 if (c.output_drive == OutputDrive::TotemPole) {
84 v |= static_cast<uint32_t>(reg::kMode2OutDrv);
85 }
86 return reg::Mode2Outne::Insert(static_cast<uint8_t>(v), OffLevelOutne(c));
87}

120[[nodiscard]] constexpr uint16_t ChannelRange(uint32_t first, uint32_t count) noexcept {
121 uint32_t mask = 0U;
122 for (uint32_t i = 0U; i < reg::kChannels; ++i) {
123 if ((i >= first) && (i < (first + count))) {
124 mask |= uint32_t{1} << i;
125 }
126 }
127 return static_cast<uint16_t>(mask);
128}
