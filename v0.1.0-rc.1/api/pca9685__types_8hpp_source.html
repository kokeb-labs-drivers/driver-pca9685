325constexpr bool IsValid(const Pwm& pwm) noexcept {
326 if ((pwm.on_count > kMaxCount) || (pwm.off_count > kMaxCount)) {

343constexpr Result<Pwm> PwmForDuty(uint16_t duty, uint16_t phase = 0U) noexcept {
344 if ((duty > kFullDuty) || (phase > kMaxCount)) {
345 return Result<Pwm>::Failure(Error::InvalidArgument);
346 }
347 if (duty == 0U) {
348 return Result<Pwm>::Success(PwmOff());
349 }
350 if (duty == kFullDuty) {
351 return Result<Pwm>::Success(PwmFullOn());
352 }
353 const uint32_t off = (static_cast<uint32_t>(phase) + static_cast<uint32_t>(duty)) & 0x0FFFU;
354 return Result<Pwm>::Success(Pwm{.on_count = phase,
355 .off_count = static_cast<uint16_t>(off),
356 .full_on = false,
357 .full_off = false});
358}

407constexpr Result<uint8_t> PrescaleFor(uint32_t frequency_hz, uint32_t oscillator_hz) noexcept {
408 if ((frequency_hz == 0U) || (frequency_hz > kMaxPwmFrequencyHz) ||
409 (oscillator_hz < kMinOscillatorHz) || (oscillator_hz > kMaxOscillatorHz)) {
410 return Result<uint8_t>::Failure(Error::InvalidArgument);
411 }
412 // round(osc / (4096 f)) as (osc + 2048 f) / (4096 f): at most 70.5e6, inside uint32_t.
413 const uint32_t divisor = kCountsPerPeriod * frequency_hz;
414 const uint32_t rounded = (oscillator_hz + (divisor / 2U)) / divisor;
415 if ((rounded < (uint32_t{reg::kPreScaleMin} + 1U)) || (rounded > 256U)) {
416 return Result<uint8_t>::Failure(Error::InvalidArgument);
417 }
418 return Result<uint8_t>::Success(static_cast<uint8_t>(rounded - 1U));
419}
420
427constexpr uint32_t FrequencyFor(uint8_t prescale, uint32_t oscillator_hz) noexcept {
428 const uint32_t divisor = kCountsPerPeriod * (static_cast<uint32_t>(prescale) + 1U);
429 return (oscillator_hz + (divisor / 2U)) / divisor;
430}

441constexpr Result<uint16_t> CountsForPulseUs(uint32_t width_us, uint8_t prescale,
442 uint32_t oscillator_hz) noexcept {
443 using detail::operator""_u16;
444 if (width_us == 0U) {
445 return Result<uint16_t>::Success(0_u16);
446 }
447 const uint64_t per_count = 1'000'000ULL * (static_cast<uint64_t>(prescale) + 1ULL);
448 const uint64_t counts =
449 ((static_cast<uint64_t>(width_us) * oscillator_hz) + (per_count / 2ULL)) / per_count;
450 if ((counts == 0ULL) || (counts > kMaxCount)) {
451 return Result<uint16_t>::Failure(Error::InvalidArgument);
452 }
453 return Result<uint16_t>::Success(static_cast<uint16_t>(counts));
454}

464constexpr uint32_t PulseNsForCounts(uint16_t counts, uint8_t prescale,
465 uint32_t oscillator_hz) noexcept {
466 const uint64_t ticks = static_cast<uint64_t>(counts) * (static_cast<uint64_t>(prescale) + 1ULL);
467 const uint64_t osc = (oscillator_hz == 0U) ? 1ULL : static_cast<uint64_t>(oscillator_hz);
468 return static_cast<uint32_t>(((ticks * 1'000'000'000ULL) + (osc / 2ULL)) / osc);
469}

504// The arithmetic and the presets, proved at build time (KB-CPP20-012).
505static_assert(Validate(Config{}) == Error::Ok);
506static_assert(Validate(DirectLedPreset()) == Error::Ok);
507static_assert(Validate(ServoPreset()) == Error::Ok);
508// Equation 2: 200 Hz from 25 MHz is 0x1E, the reset value of PRE_SCALE.
509static_assert(PrescaleFor(200U, kInternalOscillatorHz).value == reg::kPreScaleReset);
510// 1526 Hz is PRE_SCALE 3 and PRE_SCALE 255 gives 24 Hz (section 7.3.5); 1800 Hz and 23 Hz do not
511// fit.
512static_assert(PrescaleFor(1526U, kInternalOscillatorHz).value == 3U);
513static_assert(FrequencyFor(0xFFU, kInternalOscillatorHz) == 24U);
514static_assert(PrescaleFor(24U, kInternalOscillatorHz).value == 253U);
515static_assert(!PrescaleFor(1800U, kInternalOscillatorHz).ok());
516static_assert(!PrescaleFor(23U, kInternalOscillatorHz).ok());
517static_assert(!PrescaleFor(0U, kInternalOscillatorHz).ok());
518static_assert(FrequencyFor(reg::kPreScaleReset, kInternalOscillatorHz) == 197U);
519// 50 Hz: PRE_SCALE 121; 1.5 ms is 307 counts of 4.88 us.
520static_assert(PrescaleFor(50U, kInternalOscillatorHz).value == 121U);
521static_assert(CountsForPulseUs(1500U, 121U, kInternalOscillatorHz).value == 307U);
522static_assert(PulseNsForCounts(307U, 121U, kInternalOscillatorHz) == 1'498'160U);
523static_assert(!CountsForPulseUs(30'000U, 121U, kInternalOscillatorHz).ok());
524// Example 1 of section 7.3.3: a 10 % delay (409) and a 20 % duty (819) end at count 1228.
525static_assert(PwmForDuty(819U, 409U).value.off_count == 1228U);
526// Example 2: a 90 % delay (3685) and a 90 % duty (3686) wrap to 3275.
527static_assert(PwmForDuty(3686U, 3685U).value.off_count == 3275U);
528static_assert(IsOff(PwmForDuty(0U).value) && PwmForDuty(kFullDuty).value.full_on);
529static_assert(!PwmForDuty(kFullDuty + 1U).ok() && !PwmForDuty(1U, 4096U).ok());
530static_assert(IsValid(Pwm{}) && IsValid(PwmFullOn()) && !IsValid(Pwm{.full_off = false}) &&
531 !IsValid(Pwm{.on_count = 4096U}));
